\section{Proofs and Derivations}
\label{sec:appendix_proofs}

\subsection{Spherical Geometry and Detector Projection}
\label{sec:appendix_fov_projection}

Let $a=R_E+h$ and $n=\sqrt{\mu/a^3}$. A ray pitched ahead by $\alpha$ from nadir intersects the visible surface at slant range
\begin{equation}
\ell(\alpha)=a\cos\alpha-\sqrt{R_E^2-a^2\sin^2\alpha},
\label{eq:ray_slant_range}
\end{equation}
for $|\alpha|\leq\arcsin(R_E/a)$. Its signed along-track projection and central angle are
\begin{equation}
\rho(\alpha)=\ell(\alpha)\sin\alpha,
\qquad
\psi(\alpha)=\operatorname{atan2}\!\left(\rho(\alpha),a-\ell(\alpha)\cos\alpha\right).
\label{eq:ground_offset}
\end{equation}
Positive $\psi$ denotes forward lead, with $t_{\mathrm{lead}}=\psi/n$ in the circular-orbit sizing model.

Let $x$ point forward and $y$ cross-track in the plane tangent at nadir. We represent the two lateral FoR boundaries by $y=\sigma\rho_F$, where $\sigma\in\{-1,+1\}$ and $\rho_F=\rho(\theta_{\mathrm{FoR}})$. With $R_F=\sqrt{R_E^2-\rho_F^2}$, the spacecraft-to-surface depth and the visible along-track limit are
\begin{equation}
\begin{aligned}
z(x)&=a-\sqrt{R_F^2-x^2},\\
x_{\mathrm{limb}}&=\sqrt{R_E^2\left[1-(R_E/a)^2\right]-\rho_F^2},
\qquad 0\leq x\leq x_{\mathrm{limb}}.
\end{aligned}
\label{eq:main_edge_geometry}
\end{equation}
An imaginary $x_{\mathrm{limb}}$ means the FoR edge is beyond the visible surface. Within this fixed-cross-track approximation, the forward lead angle is $\psi_F(x)=\operatorname{atan2}(x,\sqrt{R_F^2-x^2})$.

For boresight pitch $b=\Theta_v/2$ and body roll $\varphi$, rotating the line of sight into the camera frame gives
\begin{equation}
\begin{aligned}
\delta_\sigma(x;\varphi)
&=\sigma\rho_F\cos\varphi-z(x)\sin\varphi,\\
\eta_\sigma(x;\varphi)
&=x\cos b-\sin b\,[z(x)\cos\varphi+\sigma\rho_F\sin\varphi],\\
d_\sigma(x;\varphi)
&=x\sin b+\cos b\,[z(x)\cos\varphi+\sigma\rho_F\sin\varphi].
\end{aligned}
\label{eq:main_detector_projection}
\end{equation}
The normalized coordinates are $u_\sigma=\delta_\sigma/d_\sigma$ and $v_\sigma=\eta_\sigma/d_\sigma$. The admissible set used in~\eqref{eq:horizontal_fov_requirement} is
\begin{equation}
\mathcal{D}_\sigma(\varphi;t_{\mathrm{req}})
=\left\{x\in[0,x_{\mathrm{limb}}]:
d_\sigma>0,\quad |v_\sigma|\leq\tan b,\quad
\psi_F(x)/n\geq t_{\mathrm{req}}\right\}.
\label{eq:admissible_fov_contact}
\end{equation}
The design maps set $t_{\mathrm{req}}=0$ to isolate geometric contact. Orbit propagation and actual access epochs supply the visibility and timing checks in the scheduler.

\subsection{Evaluation of the Edge-Contact Requirement}
\label{sec:appendix_fov_evaluation}

\paragraph{Nadir contact.}
At $\varphi=0$, the two edges are symmetric and $|u_\sigma|=\rho_F/(x\sin b+z\cos b)$ decreases with $x$. The minimum width is therefore attained at the farthest admissible point. For $\Theta_v<\ang{90}$, an upper-detector-edge contact obeys $x/z=\tan\Theta_v$, giving
\begin{equation}
x_{\mathrm{edge}}
=\sin\Theta_v\left[a\cos\Theta_v-\sqrt{R_F^2-a^2\sin^2\Theta_v}\right].
\label{eq:nadir_upper_edge_contact}
\end{equation}
Use this contact when it is real and precedes the visible limb; otherwise use the limb if it satisfies the vertical detector bound. Equivalently, the transition occurs at $\Theta_v=\operatorname{atan2}(x_{\mathrm{limb}},z(x_{\mathrm{limb}}))$. With the retained point $x_N$, the spherical nadir requirement is
\begin{equation}
\Theta_{h,\min}^{(N)}
=2\arctan\!\left[\frac{\rho_F}{x_N\sin b+z(x_N)\cos b}\right].
\label{eq:engineering_horizontal_sizing}
\end{equation}
A positive $t_{\mathrm{req}}$ additionally requires $\psi_F(x_N)/n\geq t_{\mathrm{req}}$; otherwise no earlier point can meet the lead-time requirement.

\paragraph{Rolled contact.}
For the rolled design map, the implementation samples \num{5000} uniformly spaced positions in $[0,x_{\mathrm{limb}}]$. For each edge, it projects the samples using~\eqref{eq:main_detector_projection}, retains those in $\mathcal{D}_\sigma(\varphi;0)$, and takes the smallest $|u_\sigma|$. The larger of the two edge requirements gives the horizontal width. If either edge has no retained sample, the cell is marked infeasible at this sampling resolution. This one-dimensional numerical calculation avoids separate formulas for upper-edge, limb, and interior contacts. \figref{fig:camera_fov_limit_boxes} shows representative spherical projections.

\subsection{Two-Anchor Renewal Estimator}
\label{sec:appendix_renewal}

Write $L=L_{\mathrm{anc}}$, $[\ell,r]=[\ell_\alpha,r_\alpha]$, $\lambda=\lambda_{\mathrm{c}}$, and $U=U_\alpha(g)$. After the boundary clearances in~\eqref{eq:support_clearances}--\eqref{eq:usable_support}, the first point can occur at usable-support start; each additional point needs clearance $g$ plus an exponential wait. Under Poissonization~\citep{jacquet_analytical_1998, feller_probability_1948}, placing $k$ points requires
\begin{equation}
S_k = (k{-}1)g + \underbrace{\sum_{i=1}^{k} X_i}_{Y_k},
\end{equation}
where $Y_k\sim\Gamma(k,\lambda)$~\citep{feller_introduction_1991}. Thus $k$ Poisson arrivals must fit in $U-(k{-}1)g$. Poisson--Gamma duality with $\mu=\lambda[U-(k{-}1)g]$ gives~\eqref{eq:chain_prob}. Shape $k$ counts exponential waits; anchor and internal clearances are deterministic, the two-anchor non-paralyzable (Type~I) convention~\citep{pyke_renewal_1958}.

\paragraph{Residual bias.}
\eqref{eq:chain_prob} removes only support violating anchor clearances, eliminating the dominant one-anchor boundary bias. Remaining negative bias comes from replacing $N$ fixed points by $\mathrm{Poisson}(N)$: concavity of chain length $f(n)$ gives $\mathbb{E}[f(\mathrm{Poisson}(N))]\leq f(N)$. An exact variant could average over $\mathrm{Binomial}(N_{\mathrm{obs}},p)$; we retain Poissonization for monotonicity in $g$ and algebraic pitch search. A de-Poissonization correction~\citep{jacquet_analytical_1998} adds numerical variance without improving RMSE.

\subsection{Proof of Proposition~\ref{prop:timing} (Latest Feasible Lookahead Timing)}

Decompose
$\widehat{\Delta J}(t)/\bar{c}
= E[K;\, \lambda_c(t),\, W(t),\, \ell(t),\, r(t),\, L_{\mathrm{anc}},\, g]
- p\, N_{\mathrm{sched}}(t)
- \hat{J}^-(t)/\bar{c}$,
where $E[K;\cdot]$ is the support-aware two-anchor chain-length estimator from~\eqref{eq:chain_expectation} applied to the clipped detector support $[\ell(t),r(t)]$ inside the fixed anchor interval.
Over admissible times, $N_{\mathrm{obs}}(t)$ and support end $r(t)$ are non-decreasing as the actionable window $[t_{\mathrm{next}},t+t_{\mathrm{horizon}}]$ expands, so the chain estimate cannot decrease. The gap-level baseline is fixed, and every admissible trip ends by $t_{\mathrm{next}}$, making $\hat{J}^-=0$. Hence $\widehat{\Delta J}(t)$ is non-decreasing and maximized at $t^\star(\alpha)=t_{\mathrm{next}}-t_{\mathrm{man}}(\alpha)$. \qed
